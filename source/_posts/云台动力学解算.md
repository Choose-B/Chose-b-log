J_{pitch}=J_{2-pitch}+m_2l_0^2 \\
J_{yaw}=J_{1-yaw}+J_{2-yaw}+m_2(l_0cos\phi )^2 \\
\tau_{yaw}= J_{yaw}\ddot{\theta} + \underbrace{\frac{dJ_{yaw}}{d\phi} \dot{\phi} \dot{\theta}}_{科里奥利耦合项} \\
